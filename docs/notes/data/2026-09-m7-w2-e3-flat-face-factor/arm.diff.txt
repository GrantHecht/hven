+// e3_prox_arm.h -- M7 W2 E3 SCRATCH ARM (test-only, env-toggled, NEVER shipped
+// in this form). PROXIMAL CENTRING of the primal regularization: the EQP is
+// solved with the penalty (delta/2)||x_F - x_k,F||^2 in place of the shipped
+// (delta/2)||x_F||^2, which makes the regularized answer a DESCENT direction
+// for the face's model by construction and drives the stationarity residual at
+// a converged point to -delta*p -> 0.
+//
+// Only the RIGHT-HAND SIDE changes: the stationarity row of the proximal
+// problem is (H_FF + delta I) x_F + C' lam = -g_F - shift + delta*x_k,F, so K
+// is bit-identical to the shipped assembly and the factorization is untouched.
+//
+// MODES (env E3_PROX, read once per solve):
+//   0  OFF -- the shipped behaviour: centring at 0, refinement against the
+//      FULLY unregularized system.
+//   1  proximal rhs, refinement unchanged (still against the fully
+//      unregularized system).
+//   2  proximal rhs AND refinement against the PROXIMAL system: the primal
+//      delta is part of the MODEL now, so only the dual mu is backed out.
+//   3  CONTROL: absolute centring with mode 2's refinement target, to separate
+//      the two halves of mode 2.
+
+#include <cstdlib>
+
+#include <hven/core/types.h>
+
+namespace hven::solvers::detail::e3 {
+
+inline thread_local int mode = 0;
+/// The current iterate x_k. Null means "no centre available", which falls back
+/// to the shipped absolute centring whatever `mode` says.
+inline thread_local const Vec *centre = nullptr;
+
+inline int mode_from_env() {
+    const char *v = std::getenv("E3_PROX");
+    return v != nullptr ? std::atoi(v) : 0;
+}
+
+/// True iff the rhs carries the proximal term on this call.
+inline bool prox_rhs() { return (mode == 1 || mode == 2) && centre != nullptr; }
+/// True iff refinement targets the PROXIMAL system (delta kept, dual mu out).
+inline bool prox_refine() { return mode == 2 || mode == 3; }
+
+/// Sets the centre for the duration of one walk and restores the previous one.
+class CentreGuard {
+  public:
+    explicit CentreGuard(const Vec *x) : prev_(centre), prev_mode_(mode) {
+        centre = x;
+        mode = mode_from_env();
+    }
+    CentreGuard(const CentreGuard &) = delete;
+    CentreGuard &operator=(const CentreGuard &) = delete;
+    ~CentreGuard() {
+        centre = prev_;
+        mode = prev_mode_;
+    }
+
+  private:
+    const Vec *prev_;
+    int prev_mode_;
+};

+    const QpSolution s = hven::test_support::solve_qp_dump(d, o);
+    const std::vector<detail::testing::QpWalkMinorRecord> recs = Obs::records;
+    Obs::active = false;
+    Obs::reset();
+    const double f = hven::test_support::qp_dump_objective(d, s.x);
+
+    FILE *csv = nullptr;
+    if (const char *p = std::getenv("E3_ORD"); p != nullptr) {
+        csv = std::fopen(p, "w");
+        std::fprintf(csv, "minor,action,alpha,p_inf,dx_inf,step_tol,gp,php,centre,df_full,"
+                          "df_step,gp_unref,php_unref,centre_unref,unref_dx,x_free_sq,"
+                          "eqp_free_sq,delta,mu,path,res_reg_unref,res_reg_ref,res_unreg_unref,"
+                          "res_unreg_ref,rho_unref,rho_ref,row_unref,row_ref,rhs_inf,lam_inf,"
+                          "dim,n_free,stat_free_inf,active_res,x_inf,ws_sig_in,ws_size_in\n");
+    }
+    // EVERY pass, with the ride site's own slope and sign beside the ordinary
+    // one, so the objective change over a cycle is decidable by action.
+    FILE *all = nullptr;
+    if (const char *p = std::getenv("E3_ALL"); p != nullptr) {
+        all = std::fopen(p, "w");
+        std::fprintf(all, "minor,action,alpha,p_inf,dx_inf,step_tol,gp,php,centre,ride,"
+                          "ride_sign,ride_slope,delta,x_inf,x_l1,ws_sig_in,ws_size_in,"
+                          "post_drop,rho_unref,rho_ref,res_reg_unref,row_ref\n");
+        for (const auto &r : recs) {
+            std::fprintf(all,
+                         "%ld,%d,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%d,%d,%.17g,%.17g,"
+                         "%.17g,%.17g,%llu,%ld,%d,%.17g,%.17g,%.17g,%.17g\n",
+                         static_cast<long>(r.minor), static_cast<int>(r.action), r.alpha, r.p_inf,
+                         r.dx_inf, r.step_tol, r.e3_ord_gp, r.e3_ord_php, r.e3_ord_centre,
+                         r.ride ? 1 : 0, r.ride_sign_chosen, r.ride_slope, r.primal_delta, r.x_inf,
+                         r.x_l1, static_cast<unsigned long long>(r.ws_sig_in),
+                         static_cast<long>(r.ws_size_in), r.post_drop ? 1 : 0, r.e3_rho_unref,
+                         r.e3_rho_ref, r.e3_res_reg_unref, r.e3_row_ref);
+        }
+        std::fclose(all);
+    }
+    const long tail_n =
+        std::getenv("E3_TAIL") != nullptr ? std::atol(std::getenv("E3_TAIL")) : 400L;
+    const long first_tail = static_cast<long>(recs.size()) - tail_n;
+    long ord = 0, ord_tail = 0, asc = 0, asc_tail = 0, asc_unref_tail = 0;
+    double df_sum = 0.0, df_up = 0.0;
+    std::vector<double> t_gp, t_centre, t_ratio, t_contract, t_contract_full, t_relres, t_rho_u,
+        t_rho_r, t_row_r, t_gp_unref;
+    for (const auto &r : recs) {
+        const bool ordinary = r.action == Action::kMoveOnly || r.action == Action::kAdd;
+        if (!ordinary) {
+            continue;
+        }
+        ++ord;
+        const double a = r.alpha;
+        const double df_full = r.e3_ord_gp + 0.5 * r.e3_ord_php;
+        const double df_step = a * r.e3_ord_gp + 0.5 * a * a * r.e3_ord_php;
+        df_sum += df_step;
+        if (df_step > 0.0) {
+            ++asc;
+            df_up += df_step;
+        }
+        if (csv != nullptr) {
+            std::fprintf(csv,
+                         "%ld,%d,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,"
+                         "%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%d,%.17g,%.17g,%.17g,%.17g,"
+                         "%.17g,%.17g,%.17g,%.17g,%.17g,%.17g,%ld,%ld,%.17g,%.17g,%.17g,%llu,%ld\n",
+                         static_cast<long>(r.minor), static_cast<int>(r.action), a, r.p_inf,
+                         r.dx_inf, r.step_tol, r.e3_ord_gp, r.e3_ord_php, r.e3_ord_centre, df_full,
+                         df_step, r.e3_ord_gp_unref, r.e3_ord_php_unref, r.e3_ord_centre_unref,
+                         r.e3_unref_dx, r.e3_x_free_sq, r.e3_eqp_free_sq, r.primal_delta, r.dual_mu,
+                         static_cast<int>(r.eqp_path), r.e3_res_reg_unref, r.e3_res_reg_ref,
+                         r.e3_res_unreg_unref, r.e3_res_unreg_ref, r.e3_rho_unref, r.e3_rho_ref,
+                         r.e3_row_unref, r.e3_row_ref, r.e3_rhs_inf, r.e3_lam_inf,
+                         static_cast<long>(r.e3_dim), static_cast<long>(r.e3_n_free),
+                         r.stat_free_inf, r.active_res_inf, r.x_inf,
+                         static_cast<unsigned long long>(r.ws_sig_in),
+                         static_cast<long>(r.ws_size_in));
+        }
+        if (static_cast<long>(&r - recs.data()) < first_tail) {
+            continue;
+        }
+        ++ord_tail;
+        if (df_step > 0.0) {
+            ++asc_tail;
+        }
+        if (r.e3_ord_gp_unref == r.e3_ord_gp_unref && r.e3_ord_gp_unref > 0.0) {
+            ++asc_unref_tail;
+        }
+        t_gp.push_back(r.e3_ord_gp);
+        t_gp_unref.push_back(r.e3_ord_gp_unref);
+        t_centre.push_back(r.e3_ord_centre);
+        if (r.e3_ord_centre != 0.0) {
+            t_ratio.push_back(r.e3_ord_gp / r.e3_ord_centre);
+        }
+        if (r.e3_rho_unref > 0.0) {
+            t_contract.push_back(r.e3_rho_ref / r.e3_rho_unref);
+        }
+        if (r.e3_res_unreg_unref > 0.0) {
+            t_contract_full.push_back(r.e3_res_unreg_ref / r.e3_res_unreg_unref);
+        }
+        if (r.e3_rhs_inf > 0.0) {
+            t_relres.push_back(r.e3_res_reg_ref / r.e3_rhs_inf);
+        }
+        t_rho_u.push_back(r.e3_rho_unref);
+        t_rho_r.push_back(r.e3_rho_ref);
+        t_row_r.push_back(r.e3_row_ref);
+    }
+    if (csv != nullptr) {
+        std::fclose(csv);
+    }
+    std::printf("[E3] dump=%s alg=%s status=%d minors=%ld f=%.12g records=%zu\n", dump, algs,
+                static_cast<int>(s.status), static_cast<long>(s.counters.minor_iters), f,
+                recs.size());
+    std::printf("[E3] ordinary=%ld ascending=%ld df_sum=%.6e df_up=%.6e   tail=%ld "
+                "tail_ordinary=%ld tail_ascending=%ld tail_ascending_unrefined=%ld\n",
+                ord, asc, df_sum, df_up, tail_n, ord_tail, asc_tail, asc_unref_tail);
+    std::printf("[E3] tail median: g'p=%.6e centre=%.6e g'p/centre=%.6e   g'p_unref=%.6e\n",
+                med(t_gp), med(t_centre), med(t_ratio), med(t_gp_unref));
+    std::printf("[E3] tail rho: median unref=%.6e ref=%.6e contraction(free)=%.6e "
+                "contraction(full)=%.6e   max ref=%.6e\n",
+                med(t_rho_u), med(t_rho_r), med(t_contract), med(t_contract_full), mx(t_rho_r));
+    std::printf("[E3] tail rows: median ||row resid||inf(ref)=%.6e   median "
+                "||rhs-Ky||inf/||rhs||inf(ref)=%.6e max=%.6e\n",
+                med(t_row_r), med(t_relres), mx(t_relres));
+    std::printf("[E3] counters: eqviol=%ld ws_adds=%ld ws_drops=%ld exch=%ld zmexh=%ld "
+                "predsign=%ld signv=%ld\n",
+                static_cast<long>(s.counters.eq_drift_violations),
+                static_cast<long>(s.counters.ws_adds), static_cast<long>(s.counters.ws_drops),
+                static_cast<long>(s.counters.ws_exchanges),
+                static_cast<long>(s.counters.zero_motion_exhausted),
+                static_cast<long>(s.counters.ride_predicted_signs),
+                static_cast<long>(s.counters.ride_sign_vetoes));
+    std::printf("[E3] truncated=%d allocation_failed=%d\n", Obs::truncated ? 1 : 0,
+                Obs::allocation_failed ? 1 : 0);
+}
