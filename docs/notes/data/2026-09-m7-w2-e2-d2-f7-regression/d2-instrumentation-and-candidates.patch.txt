+double zzz_feas(const QpProblem &qp, const Vec &x) {
+    double v = 0.0;
+    for (Index i = 0; i < qp.n(); ++i) v = std::max({v, qp.lower(i) - x(i), x(i) - qp.upper(i)});
+    if (qp.me() > 0) v = std::max(v, (qp.Ae * x - qp.be).lpNorm<Eigen::Infinity>());
+    if (qp.mi() > 0) v = std::max(v, (qp.Ai * x - qp.bi).maxCoeff());
+    return v;
+}
+double zzz_kkt(const QpProblem &qp, const QpSolution &s) {
+    Vec r = qp.H.selfadjointView<Eigen::Upper>() * s.x + qp.g - s.z;
+    if (qp.me() > 0) r += qp.Ae.transpose() * s.lambda_e;
+    if (qp.mi() > 0) r += qp.Ai.transpose() * s.lambda_i;
+    double v = r.lpNorm<Eigen::Infinity>();
+    for (Index j = 0; j < qp.mi(); ++j) {
+        v = std::max(v, -s.lambda_i(j));
+        v = std::max(v, std::abs(s.lambda_i(j) * (qp.bi(j) - qp.Ai.row(j).dot(s.x))));
+    }
+    for (Index i = 0; i < qp.n(); ++i) {
+        if (s.z(i) > 0) v = std::max(v, s.z(i) * std::abs(s.x(i) - qp.lower(i)));
+        if (s.z(i) < 0) v = std::max(v, -s.z(i) * std::abs(s.x(i) - qp.upper(i)));
+    }
+    return std::max(v, zzz_feas(qp, s.x));
+}

+struct ZzzGen {
+    std::uint64_t st;
+    double rnd() {
+        st = st * 6364136223846793005ull + 1442695040888963407ull;
+        return static_cast<double>((st >> 11) % 1000000) / 1000000.0;
+    }
+};
+QpProblem zzz_hand(int k) {
+    const std::vector<double> H5 = {-0.75, 0, -0.25, 0.75, 1.25, 0, 0, 0, 0, 0, -0.25, 0, 0.25, -0.25, -0.25, 0.75, 0, -0.25, 0, -0.25, 1.25, 0, -0.25, -0.25, -0.75};
+    QpProblem qp;
+    const Index n = 5;
+    Eigen::MatrixXd Hd = Eigen::Map<const Eigen::MatrixXd>(H5.data(), 5, 5);
+    Hd = Eigen::MatrixXd(Hd.selfadjointView<Eigen::Upper>());
+    if (k == 1) Hd = Eigen::MatrixXd::Identity(5, 5);
+    if (k == 2) { Hd = Eigen::MatrixXd::Identity(5, 5); Hd(1, 1) = -1; }
+    qp.H = Eigen::MatrixXd(Hd.triangularView<Eigen::Upper>()).sparseView();
+    qp.g = Vec(n); qp.g << 0.5, 0.5, 0.5, 0, -1;
+    Eigen::MatrixXd Ae = Eigen::MatrixXd::Zero(2, n);
+    Ae(0, 0) = -1; Ae(0, 3) = 1;   // -x0 + x3 = 0.5
+    Ae(1, 3) = 1; Ae(1, 2) = -1;   // x3 - x2 = 0
+    qp.Ae = Ae.sparseView(); qp.be = Vec(2); qp.be << 0.5, 0.0;
+    qp.Ai.resize(0, n); qp.bi = Vec(0);
+    qp.lower = Vec::Zero(n); qp.upper = Vec::Ones(n); qp.upper(0) = 0.0; qp.upper(2) = 0.25;
+    if (k >= 3) {
+        Ae.setZero(); Ae(0, 0) = -1; Ae(0, 2) = 1; Ae(1, 2) = 1; Ae(1, 3) = 1;
+        qp.Ae = Ae.sparseView(); qp.be << 0.5, (k == 3 ? 0.25 : 0.75);
+        qp.upper(2) = 1.0;
+        if (k == 5) { Eigen::MatrixXd Ai = Eigen::MatrixXd::Zero(1, n); Ai(0, 1) = 1; Ai(0, 4) = 1; qp.Ai = Ai.sparseView(); qp.bi = Vec(1); qp.bi << 1.5; }
+    }
+    return qp;
+}
+QpProblem zzz_gen(int trial, int family) {
+    if (family >= 10) return zzz_hand(trial);
+    ZzzGen G{0x9E3779B97F4A7C15ull + 7919ull * static_cast<std::uint64_t>(trial) +
+             104729ull * static_cast<std::uint64_t>(family)};
+    for (int k = 0; k < 5; ++k) G.rnd();
+    const Index n = 3 + static_cast<Index>(G.rnd() * 3);
+    const Index mi = static_cast<Index>(G.rnd() * 4);
+    const Index me = 1 + static_cast<Index>(G.rnd() * (family >= 3 ? 3 : 2));
+    Eigen::VectorXd l(n), l2(n);
+    for (Index i = 0; i < n; ++i) l(i) = G.rnd() < 0.4 ? 0.0 : std::round(4 * G.rnd() - 2) / 2;
+    for (Index i = 0; i < n; ++i) l2(i) = G.rnd() < 0.5 ? 0.0 : std::round(4 * G.rnd() - 2) / 2;
+    Eigen::MatrixXd Hd = l * l.transpose();
+    if (family >= 1) Hd -= l2 * l2.transpose();
+    QpProblem qp;
+    qp.H = Eigen::MatrixXd(Hd.triangularView<Eigen::Upper>()).sparseView();
+    qp.g = Vec(n);
+    for (Index i = 0; i < n; ++i) qp.g(i) = std::round(8 * G.rnd() - 4) / 2;
+    Eigen::MatrixXd Ae = Eigen::MatrixXd::Zero(me, n), Ai = Eigen::MatrixXd::Zero(mi, n);
+    qp.be = Vec(me);
+    qp.bi = Vec(mi);
+    for (Index r = 0; r < me; ++r) {
+        const Index a = static_cast<Index>(G.rnd() * n), b = (a + 1 + static_cast<Index>(G.rnd() * (n - 1))) % n;
+        Ae(r, a) = 1; Ae(r, b) = G.rnd() < 0.5 ? 1 : -1;
+        if (family >= 3) Ae(r, static_cast<Index>(G.rnd() * n)) += G.rnd() < 0.5 ? 1 : -1;
+        qp.be(r) = std::round(6 * G.rnd() - 3) / 2;
+    }
+    for (Index r = 0; r < mi; ++r) {
+        const Index a = static_cast<Index>(G.rnd() * n), b = (a + 1 + static_cast<Index>(G.rnd() * (n - 1))) % n;
+        Ai(r, a) = G.rnd() < 0.5 ? 1 : -1; Ai(r, b) = G.rnd() < 0.5 ? 1 : -1;
+        qp.bi(r) = std::round(6 * G.rnd() - 3) / 2;
+    }
+    qp.Ae = Ae.sparseView();
+    qp.Ai = Ai.sparseView();
+    const double lo = G.rnd() < 0.5 ? 0.0 : -1.0;
+    qp.lower = Vec::Constant(n, lo);
+    qp.upper = Vec::Constant(n, 1.0);
+    if (family >= 2 && G.rnd() < 0.7) { qp.lower(0) = 0.0; qp.upper(0) = 0.0; }
+    return qp;
+}
+void zzz_print(const QpProblem &qp) {
+    Eigen::IOFormat f(Eigen::FullPrecision, 0, ", ", ";\n", "[", "]");
+    std::cout << "H=\n" << Eigen::MatrixXd(qp.H) << "\ng=" << qp.g.transpose() << "\nAe=\n" << Eigen::MatrixXd(qp.Ae)

+              << "\nlo=" << qp.lower.transpose() << "\nhi=" << qp.upper.transpose() << "\n";
+}
