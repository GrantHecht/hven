#!/usr/bin/env bash
# usage: cell.sh OUTPREFIX PROBLEM LABEL [env...]
set -o pipefail
out="$1"; prob="$2"; label="$3"; shift 3
cd /home/ghecht/Projects/hven-bench || exit 3
env MKL_NUM_THREADS=1 OMP_NUM_THREADS=1 "$@" timeout -s KILL 580 /home/ghecht/Projects/hven/.scratch/m7w2e2d1/bench-build/lanes/hven/hven_bench_cell \
  --lane hven-sqp --problem "$prob" --options "$label" --budget-iters 100 --options-dump "${out}.dump" --assert-reference > "${out}.out" 2> "${out}.err"
rc=$?; echo "cell rc=$rc"; return $rc 2>/dev/null || exit $rc
