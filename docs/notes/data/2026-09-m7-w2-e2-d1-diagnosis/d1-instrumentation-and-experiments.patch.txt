+// KKT of the QP at x with the returned lambda_e, lambda_i, over the TR-tightened
+// box [max(l, -D), min(u, D)] (window centred at 0: the driver zeroes seed.x).
+// The bound multiplier is the BEST one: z_i = r_i where r = Hx+g+Ae'le+Ai'li,
+// with the sign condition at an active bound, |r_i| on a free coordinate.
+void d1_kkt(const QpProblem &qp, const QpSolution &s, double D, double tol, const char *lab) {
+    const Index n = qp.n();
+    if (s.x.size() != n) { std::fprintf(stderr, "D1KKT %s no x\n", lab); return; }
+    Vec lo = qp.lower, up = qp.upper;
+    if (std::isfinite(D)) { for (Index i = 0; i < n; ++i) { lo(i) = std::max(lo(i), -D); up(i) = std::min(up(i), D); } }
+    const Vec Hx = qp.H.selfadjointView<Eigen::Upper>() * s.x;
+    Vec r = Hx + qp.g;
+    const bool have_mult = s.lambda_i.size() == qp.mi() && s.lambda_e.size() == qp.me();
+    if (have_mult) {
+        if (qp.me() > 0) r += qp.Ae.transpose() * s.lambda_e;
+        if (qp.mi() > 0) r += qp.Ai.transpose() * s.lambda_i;
+    }
+    double stat = 0, bviol = 0, eqviol = 0, iviol = 0, lneg = 0, comp = 0, gscale = 1.0;
+    gscale = std::max(1.0, qp.g.lpNorm<Eigen::Infinity>());
+    Index nlo = 0, nup = 0;
+    for (Index i = 0; i < n; ++i) {
+        const double xi = s.x(i);
+        bviol = std::max({bviol, lo(i) - xi, xi - up(i)});
+        const bool atl = lo(i) > -1e19 && xi - lo(i) <= 1e-9 * std::max(1.0, std::abs(lo(i)));
+        const bool atu = up(i) < 1e19 && up(i) - xi <= 1e-9 * std::max(1.0, std::abs(up(i)));
+        if (atl && atu) continue;
+        if (atl) { ++nlo; stat = std::max(stat, -r(i)); }
+        else if (atu) { ++nup; stat = std::max(stat, r(i)); }
+        else stat = std::max(stat, std::abs(r(i)));
+    }
+    if (qp.me() > 0) eqviol = (qp.Ae * s.x - qp.be).lpNorm<Eigen::Infinity>();
+    if (qp.mi() > 0) {
+        const Vec ax = qp.Ai * s.x;
+        for (Index j = 0; j < qp.mi(); ++j) {
+            iviol = std::max(iviol, ax(j) - qp.bi(j));
+            if (have_mult) { lneg = std::max(lneg, -s.lambda_i(j)); comp = std::max(comp, std::abs(s.lambda_i(j) * std::min(0.0, ax(j) - qp.bi(j)))); }
+        }
+    }
+    const double obj = qp.g.dot(s.x) + 0.5 * s.x.dot(Hx);
+    std::fprintf(stderr,
+                 "D1KKT %s status=%s obj=%.12e stat=%.3e(scale %.3e) eqviol=%.3e iviol=%.3e boxviol=%.3e lneg=%.3e comp=%.3e mult=%d atlo=%ld atup=%ld |x|inf=%.6e KKT@tol%.0e=%s\n",
+                 lab, d1_status(s.status), obj, stat, gscale, eqviol, iviol, bviol, lneg, comp, have_mult ? 1 : 0,
+                 (long)nlo, (long)nup, s.x.lpNorm<Eigen::Infinity>(), tol,
+                 (have_mult && stat <= tol * gscale && eqviol <= 1e-6 && iviol <= 1e-6 && bviol <= 1e-6 && lneg <= tol * gscale) ? "YES" : "NO");
+}
